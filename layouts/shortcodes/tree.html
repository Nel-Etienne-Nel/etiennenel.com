{{- /* File tree. Write the tree with its box-drawing guides, one entry per line: {{< tree title="home-lab/" >}} home-lab/ ├── CLAUDE.md # the schema ├── 00 - Hardware/ (+) └── log.md (M) {{< /tree >}} Folders end in "/" and render bold. "# note" goes to the right-hand column. A trailing (+) new, (M) changed or (-) deleted fills the status column. */ -}} {{- $rows := slice -}} {{- $hasStatus := false -}} {{- range split (strings.TrimSpace .Inner) "\n" -}} {{- $line := strings.TrimRight " \t\r" . -}} {{- $note := "" -}} {{- with findRESubmatch `^(.*?)\s+#\s+(.*)$` $line 1 }}{{ $line = index (index . 0) 1 }}{{ $note = index (index . 0) 2 }}{{ end -}} {{- $status := "" -}} {{- with findRESubmatch `^(.*?)\s*\((\+|M|-|−)\)$` $line 1 }}{{ $line = index (index . 0) 1 }}{{ $status = index (index . 0) 2 }}{{ $hasStatus = true }}{{ end -}} {{- $guide := index (findRE `^[│├└─ ]*` $line 1) 0 | default "" -}} {{- $name := strings.TrimPrefix $guide $line -}} {{- $rows = $rows | append (dict "guide" $guide "name" $name "note" $note "status" (replace $status "-" "−") "dir" (hasSuffix $name "/")) -}} {{- end -}} {{- /* Emitted as one line: inside a list (e.g. steps) multi-line HTML would be cut short by Markdown. */ -}} {{- partial "tree-block.html" (dict "rows" $rows "hasStatus" $hasStatus "title" (.Get "title")) | replaceRE `>\s*\n\s*<` "><" | safeHTML -}}
