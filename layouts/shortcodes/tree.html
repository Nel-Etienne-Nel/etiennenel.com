"# note" goes to the right-hand column. A trailing (+) new, (M) changed or (-) deleted fills the status column. */ -}} {{- $rows := slice -}} {{- $hasStatus := false -}} {{- range split (strings.TrimSpace .Inner) "\n" -}} {{- $line := strings.TrimRight " \t\r" . -}} {{- $note := "" -}} {{- with findRESubmatch `^(.*?)\s+#\s+(.*)$` $line 1 }}{{ $line = index (index . 0) 1 }}{{ $note = index (index . 0) 2 }}{{ end -}} {{- $status := "" -}} {{- with findRESubmatch `^(.*?)\s*\((\+|M|-|−)\)$` $line 1 }}{{ $line = index (index . 0) 1 }}{{ $status = index (index . 0) 2 }}{{ $hasStatus = true }}{{ end -}} {{- $guide := index (findRE `^[│├└─ ]*` $line 1) 0 | default "" -}} {{- $name := strings.TrimPrefix $guide $line -}} {{- $rows = $rows | append (dict "guide" $guide "name" $name "note" $note "status" (replace $status "-" "−") "dir" (hasSuffix $name "/")) -}} {{- end -}}
